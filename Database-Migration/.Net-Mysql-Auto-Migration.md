# Auto Migration .Net and MySQL

```sh


#!/usr/bin/env bash
set -e

INFRA="Project.Infrastructure"
API="Project.Api"
STARTUP="$API/$API.csproj"
MIGRATION_PROJECT="$INFRA/$INFRA.csproj"

echo "🔍 Checking EF Core model changes..."

if dotnet ef migrations has-pending-model-changes \
    --project "$MIGRATION_PROJECT" \
    --startup-project "$STARTUP" \
    > /dev/null; then

    echo "❌ No model changes detected."

else
    echo "💡 Model changes detected. Creating migration..."

    MIGRATION_NAME="Auto_$(date +%Y_%m_%d_%H_%M_%S)"

    dotnet ef migrations add "$MIGRATION_NAME" \
        --project "$MIGRATION_PROJECT" \
        --startup-project "$STARTUP" \
        > /dev/null
fi

echo "🚀 Applying migrations to database..."

dotnet ef database update \
    --project "$MIGRATION_PROJECT" \
    --startup-project "$STARTUP" \
    > /dev/null

echo "✅ Migration workflow completed!"


```
